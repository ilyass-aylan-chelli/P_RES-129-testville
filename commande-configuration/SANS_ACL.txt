! ==========================================
! SWITCH 1 - Bâtiment Bibliothèque / Guest
! ==========================================
enable 
conf t

! Création des VLANs
vlan 20
 name bibliothèque
exit

vlan 70
 name guest
exit

! Configuration des ports d'accès
interface FastEthernet 0/1
 switchport mode access
 switchport access vlan 70
exit

interface FastEthernet 0/2
 switchport mode access
 switchport access vlan 20
exit

interface FastEthernet 0/10
 switchport mode access
 switchport access vlan 20
exit

interface FastEthernet 0/23
 switchport mode access
 switchport access vlan 70
exit

interface FastEthernet 0/24
 switchport mode access
 switchport access vlan 70
exit

! Trunk vers Routeur 1
interface FastEthernet 0/2
switchport mode access
switchport access vlan 20
exit


! ==========================================
! SWITCH 2 - Direction & Finance
! ==========================================

vlan 30

 switchport mode trunk
 switchport trunk allowed vlan 50,30

! ==========================================
! ROUTEUR 1 - Site Bibliothèque
! ==========================================
enable
configure terminal

! Liaison point-à-point vers Routeur 2
interface gigabitEthernet 0/0
 ip address 10.0.0.1 255.255.255.252
 no shutdown
exit

! Interface physique Trunk
interface gigabitEthernet 0/1
 no shutdown
exit

! Routage Inter-VLAN (Router-on-a-stick)
interface gigabitEthernet 0/1.20
 encapsulation dot1Q 20
 ip address 192.168.20.1 255.255.255.240
exit

interface gigabitEthernet 0/1.70
 encapsulation dot1Q 70
 ip address 192.168.70.1 255.255.255.224
exit 

! Routes statiques vers les autres réseaux via R2
ip route 192.168.10.0 255.255.255.248 10.0.0.2
ip route 192.168.30.0 255.255.255.192 10.0.0.2
ip route 192.168.40.0 255.255.255.192 10.0.0.2
ip route 192.168.50.0 255.255.255.192 10.0.0.2
ip route 192.168.60.0 255.255.255.224 10.0.0.2

! Agent relais DHCP vers le serveur central (CDC)
interface gigabitEthernet 0/1.20
 ip helper-address 192.168.10.3
exit

interface gigabitEthernet 0/1.70
 ip helper-address 192.168.10.3
exit

end
write memory


! ==========================================
! ROUTEUR 2 - Cœur Bâtiment Central
! ==========================================
enable
configure terminal

! Interconnexion Point-à-Point R2 <-> R1
interface gigabitEthernet 0/0
 ip address 10.0.0.2 255.255.255.252
 no shutdown
exit

! Interconnexion Point-à-Point R2 <-> R3
interface gigabitEthernet 2/0
 ip address 10.0.0.5 255.255.255.252
 no shutdown
exit

! Inter-VLAN Direction & Finance (Switch 2)
interface gigabitEthernet 1/0
 no shutdown
exit

interface gigabitEthernet 1/0.50
 encapsulation dot1Q 50
 ip address 192.168.50.1 255.255.255.192
exit

interface gigabitEthernet 1/0.30
 encapsulation dot1Q 30
 ip address 192.168.30.1 255.255.255.192
exit

! Inter-VLAN Accueil & Serveurs CDC (Switch 3)
interface gigabitEthernet 3/0
 no shutdown
exit

interface gigabitEthernet 3/0.40
 encapsulation dot1Q 40
 ip address 192.168.40.1 255.255.255.192
exit

interface gigabitEthernet 3/0.10
 encapsulation dot1Q 10
 ip address 192.168.10.1 255.255.255.248
exit

! Routes statiques vers R1 et R3
ip route 192.168.20.0 255.255.255.240 10.0.0.1
ip route 192.168.70.0 255.255.255.224 10.0.0.1
ip route 192.168.60.0 255.255.255.224 10.0.0.6

! Agent relais DHCP vers le serveur central
interface gigabitEthernet 1/0.50
 ip helper-address 192.168.10.3
exit

interface gigabitEthernet 1/0.30
 ip helper-address 192.168.10.3
exit

interface gigabitEthernet 3/0.40
 ip helper-address 192.168.10.3
exit

end
write memory
